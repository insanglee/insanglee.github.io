parse_books <- function(pg) {
  pods <- html_elements(pg, "article.product_pod")
  
  tibble(
    title = pods |> html_element("h3 a") |> html_attr("title"),
    price = pods |> html_element(".price_color") |> html_text2(),
    rating = pods |> html_element(".star-rating") |> html_attr("class"),
    stock = pods |> html_element(".availability") |> html_text2()
  )
}

stars <- c("One", "Two", "Three", "Four", "Five")

clean_books <- function(df) {
  df |>
    mutate(
      price = parse_number(price),
      rating = match(str_remove(rating, "star-rating "), stars),
      in_stock = str_detect(stock, "In stock")
    ) |>
    select(-stock)
}

B2. Find and Call the API

library(httr2)

ua <- "EPPS6302 student; ixl240000@utdallas.edu"

get_quotes <- function(page) {
  Sys.sleep(1)
  
  request("https://quotes.toscrape.com/api/quotes") |>
    req_url_query(page = page) |>
    req_user_agent(ua) |>
    req_retry(max_tries = 3) |>
    req_perform() |>
    resp_body_json()
}

B3. Rectangle the JSON

rectangle <- function(quote_list) {
  tibble(q = quote_list) |>
    unnest_wider(q) |>
    hoist(
      author,
      author_name = "name",
      author_slug = "slug"
    ) |>
    mutate(
      tags = map_chr(
        tags,
        \(x) paste(unlist(x), collapse = "; ")
      )
    ) |>
    select(
      text,
      author_name,
      author_slug,
      tags
    )
}

Data Notes

Books to Scrape

  • Source and date collected: Books to Scrape, collected October 7, 2026.
  • Permission: The target path was checked with robotstxt, and the site is intended as a scraping sandbox.
  • Method: Data were collected with a polite session using HTML/CSS selectors and a five-second crawl delay.
  • Cleaning: Prices were converted to numeric values, ratings to integers from 1 to 5, and stock status to logical values.
  • Known gaps: The scraper depends on the current HTML structure and CSS class names, so a website redesign could break the selectors.

Quotes to Scrape

  • Source and date collected: Quotes to Scrape, collected October 7, 2026.
  • Permission: The site is designed for web-scraping practice, and requests were limited to one per second.
  • Method: Quotes were collected from the site’s API and converted from JSON into a structured tibble; the JavaScript page was also tested with a live browser.
  • Cleaning: Nested author and tag information was converted into one row per quote with text, author name, author slug, and tags.
  • Known gaps: The process depends on the API endpoint and response structure remaining available and unchanged.

YouTube Data API

  • Source and date collected: YouTube Data API v3, collected October 7, 2026.
  • Permission: Access was made through a personal API key stored securely in .Renviron.
  • Method: The documented YouTube Data API was queried for the ten most popular videos in the United States.
  • Cleaning: Nested JSON fields were converted into a table containing video ID, title, channel, and view count.
  • Known gaps: The results represent only the ten most popular videos returned at the time of collection, and view counts can change over time.
